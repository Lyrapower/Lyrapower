## ref 变量与引用
%%
a = [1, 2]
b = a
b.append(3)
print(a)
= [1, 2, 3]
? [1, 2] || [3]
! b = a 只是多一个名字,指向同一个列表。
%%
a = [1, 2]
b = a[:]
b.append(3)
print(a)
= [1, 2]
? [1, 2, 3] || [3]
! a[:] 切出一份新列表,改 b 不影响 a。
%%
a = 5
b = a
b += 1
print(a)
= 5
? 6 || 1
! int 不可变,b += 1 让 b 指向新对象。
%%
x = [1, 2]
y = [1, 2]
print(x == y, x is y)
= True False
? True True || False False
! == 比内容,is 比是不是同一个对象。
%%
a = [[0]] * 3
a[0].append(1)
print(a)
= [[0, 1], [0, 1], [0, 1]]
? [[0, 1], [0], [0]] || [[0], [0], [0], 1]
! 乘出来的三格是同一个内层列表。
%%
a = [[1], [2]]
b = a.copy()
b[0].append(9)
print(a)
= [[1, 9], [2]]
? [[1], [2]] || [[1], [2], 9]
! copy() 只复制外层,内层列表仍是共用的。
%%
import copy
a = [[1], [2]]
b = copy.deepcopy(a)
b[0].append(9)
print(a)
= [[1], [2]]
? [[1, 9], [2]] || [[9], [2]]
! deepcopy 连内层一起复制。
%%
a, b = 1, 2
a, b = b, a
print(a, b)
= 2 1
? 1 2 || 2 2
! 右边先打包成元组,再一起解包,所以能直接交换。
%%
def f(x):
    x.append(1)
lst = []
f(lst)
print(lst)
= [1]
? [] || None
! 传进去的是同一个列表,原地修改外面看得见。
%%
def f(x):
    x = x + [1]
lst = []
f(lst)
print(lst)
= []
? [1] || None
! x + [1] 造了新列表并只改了函数里的名字。
%%
a = [1, 2, 3]
b = a
a = a + [4]
print(b)
= [1, 2, 3]
? [1, 2, 3, 4] || [4]
! a + [4] 是新列表;b 还指着旧的。
%%
a = [1, 2, 3]
b = a
a += [4]
print(b)
= [1, 2, 3, 4]
? [1, 2, 3] || [4]
! 列表的 += 是原地扩展,和 a = a + [4] 不同。
%%
s = 'abc'
t = s
t += 'd'
print(s)
= abc
? abcd || d
! 字符串不可变,+= 生成新串。
%%
x = None
print(x is None, x == False)
= True False
? True True || False False
! None 不等于 False;判空用 is None。

? False
! 浮点比较的正确姿势。

? 43 43 || 421 421
! 类型决定 + 是加法还是拼接。
%%

= 3 4
? 4 4 || 3 3
! int() 直接截断小数,不四舍五入。

? 5

%%

%%

! bool 是 int 的子类,True 就是 1。
%%

! :, 加千分位。
%%

? 0.3% || 25.6
! % 格式先乘 100 再加百分号。
%%

= True
? False || OverflowError
! inf 比任何有限数都大。
%%
print(bool(0), bool(0.0), bool(-1))

? False False False || True True True
! 只有零值是假,负数也是真。
%%
print(1_000_000 + 1)

? SyntaxError || 1_000_001
! 数字里的下划线只是给人看的分隔符。
%%

= 3 5 6
? -3 5 6 || 3 2 6
! 三个最常用的内置函数。
%%

= ValueError
? 3 || 4
! int() 不吃带小数点的字符串;先 float 再 int。
%%

? int int || float float
! / 得 float,// 两边都是 int 时得 int。

%%

%%

print(df.a.rank().tolist(), df.a.idxmax())
= [3.0, 1.0, 2.0] 0
? [1.0, 2.0, 3.0] 0 || [3.0, 1.0, 2.0] 3
! rank 给名次;idxmax 给最大值所在的行标签。
